import json
import runpy
import tempfile
from pathlib import Path
import torch

torch.set_num_threads(1)
base = (Path(__file__).resolve().parents[4] / '09-split-k')
u = runpy.run_path(str(base / 'tune.py'))
shape = (16, 4096, 8192)
u['inputs'].__globals__['SHAPE'] = shape
data = u['inputs'](13)
rows = []
for local in (False, True):
    source = (base / 'solution.py').read_text()
    if local:
        source = source.replace('fx.barrier()', 'fx.barrier(address_spaces=ir.ArrayAttr.get([ir.Attribute.parse("#gpu.address_space<workgroup>")]))')
    with tempfile.NamedTemporaryFile(mode='w', suffix='.py', prefix=f'_local_barrier_{local}_', dir=base) as f:
        f.write(source)
        f.flush()
        m = runpy.run_path(f.name)
        config = m['CONFIGS_BY_SHAPE'][shape]
        launch = m['launch_for'](config, shape)
        u['validate'](launch, data)
        m['check_boundaries'](config)
        report = u['paired_measure'](launch, u['torch_launch'], data, 7, 20, 0, 'graph-host')
        rows.append({'local': local, **report})
        print(local, round(report['kernel_ms'] * 1000, 3), round(report['torch_ms'] * 1000, 3), round(report['paired_ratio'], 4), flush=True)
Path('/tmp/skinny-local-barrier-probe.json').write_text(json.dumps(rows, indent=2) + '\n')
