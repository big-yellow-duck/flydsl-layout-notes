import json
import runpy
import tempfile
from pathlib import Path
import torch

torch.set_num_threads(1)
base=(Path(__file__).resolve().parents[4] / '09-split-k')
u=runpy.run_path(str(base/'tune.py')); shape=(16,4096,8192)
u['inputs'].__globals__['SHAPE']=shape
data=u['inputs'](13);rows=[]
for cache in (0,2):
 for rotation in (0,1,2,4,8,16,32,64):
  source=(base/'solution.py').read_text()
  source=source.replace('def gmem2reg_load_tile(tile_k):',f'def gmem2reg_load_tile(tile_k):\n            tile_k = initial_tile_k + (tile_k - initial_tile_k + block_col * {rotation}) % tiles_per_split')
  source=source.replace('fx.barrier()', 'fx.barrier(address_spaces=ir.ArrayAttr.get([ir.Attribute.parse("#gpu.address_space<workgroup>")]))')
  source=source.replace('gmem_load_128b = fx.make_copy_atom(fx.rocdl.BufferCopy128b(), fx.BFloat16)',f'gmem_left_load = fx.make_copy_atom(fx.rocdl.BufferCopy128b(), fx.BFloat16)\n        gmem_right_load = fx.make_copy_atom(fx.rocdl.BufferCopy128b({cache}), fx.BFloat16)')
  source=source.replace('fx.copy(gmem_load_128b, left_lds_source','fx.copy(gmem_left_load, left_lds_source').replace('fx.copy(gmem_load_128b, right_lds_source','fx.copy(gmem_right_load, right_lds_source')
  with tempfile.NamedTemporaryFile(mode='w',suffix='.py',prefix='_k_rotation_',dir=base) as f:
   f.write(source);f.flush();m=runpy.run_path(f.name)
   for bn in (32,64):
    c=m['Config'](16,bn,16,1,1,split_k=1)
    launch=m['launch_for'](c,shape);u['validate'](launch,data)
    r=u['paired_measure'](launch,u['torch_launch'],data,3,5,0,'graph-host')
    rows.append({'cache':cache,'rotation':rotation,'bn':bn,**r})
    print(cache,rotation,bn,round(r['kernel_ms']*1000,3),round(r['torch_ms']*1000,3),round(r['paired_ratio'],4),flush=True)
Path('/tmp/skinny-k-rotation-probe.json').write_text(json.dumps(rows,indent=2)+'\n')
