import json
import runpy
import tempfile
from pathlib import Path
import torch

torch.set_num_threads(1)
base = (Path(__file__).resolve().parents[4] / '09-split-k')
u = runpy.run_path(str(base / 'tune.py'))
shape = (16, 4096, 8192)
u['inputs'].__globals__['SHAPE'] = shape
data = u['inputs'](13)
rows=[]
for pad in (0,8):
 for cache in (0,2):
  source=(base/'solution.py').read_text()
  source=source.replace('BLOCK_K * BLOCK_N, 16]',f'BLOCK_N * (BLOCK_K + {pad}), 16]')
  source=source.replace('fx.make_layout((BLOCK_K, BLOCK_N), (BLOCK_N, 1))',f'fx.make_layout((BLOCK_K, BLOCK_N), (1, BLOCK_K + {pad}))')
  source=source.replace('fx.make_layout((BLOCK_N, BLOCK_K), (1, BLOCK_N))',f'fx.make_layout((BLOCK_N, BLOCK_K), (BLOCK_K + {pad}, 1))')
  source=source.replace('right_copy_plan = fx.make_tiled_copy(copy_128b,', 'right_copy_plan = fx.make_tiled_copy(copy_16b_lds,')
  source=source.replace('copy_128b = fx.make_copy_atom(fx.UniversalCopy128b(), fx.BFloat16)', 'copy_128b = fx.make_copy_atom(fx.UniversalCopy128b(), fx.BFloat16)\n        copy_16b_lds = fx.make_copy_atom(fx.UniversalCopy16b(), fx.BFloat16)')
  source=source.replace('fx.copy(copy_128b, right_fragment,', 'fx.copy(copy_16b_lds, right_fragment,')
  source=source.replace('gmem_load_128b = fx.make_copy_atom(fx.rocdl.BufferCopy128b(), fx.BFloat16)',f'gmem_left_load = fx.make_copy_atom(fx.rocdl.BufferCopy128b(), fx.BFloat16)\n        gmem_right_load = fx.make_copy_atom(fx.rocdl.BufferCopy128b({cache}), fx.BFloat16)')
  source=source.replace('fx.copy(gmem_load_128b, left_lds_source','fx.copy(gmem_left_load, left_lds_source').replace('fx.copy(gmem_load_128b, right_lds_source','fx.copy(gmem_right_load, right_lds_source')
  source=source.replace('fx.barrier()', 'fx.barrier(address_spaces=ir.ArrayAttr.get([ir.Attribute.parse("#gpu.address_space<workgroup>")]))')
  with tempfile.NamedTemporaryFile(mode='w',suffix='.py',prefix=f'_b_transpose_{pad}_{cache}_',dir=base) as f:
   f.write(source);f.flush()
   m=runpy.run_path(f.name)
   for bn,bk,wn in ((16,16,1),(32,16,1),(64,16,1),(32,32,2),(64,32,2),(64,64,2)):
    try:
     config=m['Config'](16,bn,bk,1,wn,split_k=1)
     launch=m['launch_for'](config,shape);u['validate'](launch,data)
     r=u['paired_measure'](launch,u['torch_launch'],data,3,5,0,'graph-host')
     rows.append({'pad':pad,'cache':cache,'bn':bn,'bk':bk,'wn':wn,**r})
     print(pad,cache,bn,bk,wn,round(r['kernel_ms']*1000,3),round(r['torch_ms']*1000,3),round(r['paired_ratio'],4),flush=True)
    except (ValueError,RuntimeError,AssertionError) as error:
     print('FAILED',pad,cache,bn,bk,wn,str(error)[:2000],flush=True)
Path('/tmp/skinny-b-transpose-probe.json').write_text(json.dumps(rows,indent=2)+'\n')
