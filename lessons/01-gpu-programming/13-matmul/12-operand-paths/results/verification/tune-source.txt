"""Compare mixed operand paths with tuned prefetch, dedicated waves, and Torch."""

import argparse
import hashlib
import inspect
import json
import runpy
import statistics
from dataclasses import asdict
from pathlib import Path

HERE = Path(__file__).resolve().parent
MODULE = runpy.run_path(str(HERE / "solution.py"))
HARNESS = runpy.run_path(str(HERE.parent / "11-wave-specialization/tune.py"))
COMMON = HARNESS["COMMON"]
PREFETCH = HARNESS["PREFETCH"]
SPECIALIZED = HARNESS["MODULE"]
SELECTION = json.loads(
    (HERE.parent / "11-wave-specialization/results/selection.json").read_text()
)


def configure(shape):
    COMMON["inputs"].__globals__["SHAPE"] = tuple(shape)


def controls(shape, config):
    key = "x".join(map(str, shape))
    tuned = PREFETCH["Config"](**SELECTION["prefetch_control"][key]["config"])
    launches = {
        "prefetch": PREFETCH["launch_for"](tuned, shape),
        "wave_specialized": SPECIALIZED["launch_for"](
            SPECIALIZED["CONFIGS_BY_SHAPE"][shape], shape
        ),
        "torch": COMMON["torch_launch"],
    }
    compatible = {
        name: value
        for name, value in asdict(config).items()
        if name in PREFETCH["Config"].__dataclass_fields__
    }
    try:
        matched = PREFETCH["Config"](**compatible)
        PREFETCH["validate_config"](matched, shape)
    except ValueError:
        pass
    else:
        launches["matched_prefetch"] = PREFETCH["launch_for"](matched, shape)
    return launches


def environment(shape):
    configure(shape)
    factory_hash = hashlib.sha256(
        inspect.getsource(MODULE["build"].__wrapped__).encode()
    ).hexdigest()
    return {
        **COMMON["environment"](),
        "kernel_build_sha256": factory_hash,
        "native_factory_sha256": factory_hash,
        "source_sha256": {
            name: hashlib.sha256((HERE / name).read_bytes()).hexdigest()
            for name in ("solution.py", "tune.py")
        },
        "controls": {
            "prefetch": SELECTION["prefetch_control"]["x".join(map(str, shape))][
                "config"
            ],
            "wave_specialized": asdict(SPECIALIZED["CONFIGS_BY_SHAPE"][shape]),
        },
        "helper_sha256": {
            str(path.relative_to(HERE.parents[3])): hashlib.sha256(
                path.read_bytes()
            ).hexdigest()
            for path in (
                HERE.parent / "09-split-k/tune.py",
                HERE.parent / "10-wave-local-tiles/solution.py",
                HERE.parent / "11-wave-specialization/solution.py",
                HERE.parent / "11-wave-specialization/tune.py",
            )
        },
    }


def compare(config, shape, timing="graph-host", launch_factory=None):
    shape = tuple(shape)
    configure(shape)
    launch_factory = MODULE["launch_for"] if launch_factory is None else launch_factory
    data = COMMON["inputs"](13)
    launches = {"specialized": launch_factory(config, shape), **controls(shape, config)}
    for launch in launches.values():
        COMMON["validate"](launch, data)
    result = HARNESS["measure"](launches, data, 7, 10, timing=timing)
    return {
        **result,
        "kernel_ms": result["median_ms"]["specialized"],
        "torch_ms": result["median_ms"]["torch"],
        "paired_ratio": result["ratio_to_torch"],
    }


def verify(config, shape, seeds, timing):
    configure(shape)
    result = {
        "kind": "held_out_verification",
        "config": asdict(config),
        "environment": environment(shape),
        "seeds": seeds,
        "timing": timing,
        "rounds": [],
    }
    launches = {
        "specialized": MODULE["launch_for"](config, shape),
        **controls(shape, config),
    }
    for seed in seeds:
        data = COMMON["inputs"](seed)
        for launch in launches.values():
            COMMON["validate"](launch, data)
        for repetition in range(3):
            measured = HARNESS["measure"](
                launches, data, 7, 20, seed + repetition, timing=timing
            )
            result["rounds"].append(
                {"seed": seed, "repetition": repetition, **measured}
            )
            print("VERIFY", seed, repetition, measured["median_ms"], flush=True)
    result["median_ms"] = {
        name: statistics.median(row["median_ms"][name] for row in result["rounds"])
        for name in launches
    }
    result["comparisons"] = {
        name: {
            "median_paired_ratio": statistics.median(
                row["ratio_to_" + name] for row in result["rounds"]
            ),
            "worst_round_ratio": max(
                row["ratio_to_" + name] for row in result["rounds"]
            ),
            "win_rounds": sum(
                row["ratio_to_" + name] < 0.95 for row in result["rounds"]
            ),
            "rounds": len(result["rounds"]),
        }
        for name in launches
        if name != "specialized"
    }
    result["criterion"] = (
        "Win: every round's median paired ratio <0.95; parity: every round <=1.05."
    )
    return result


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--shape", type=int, nargs=3, default=(32, 1024, 8192))
    parser.add_argument("--config", help="Explicit JSON configuration object")
    parser.add_argument(
        "--timing", choices=("graph-host", "direct-host"), default="graph-host"
    )
    parser.add_argument(
        "--seeds", type=int, nargs="+", default=(2026112601, 2026112602, 2026112603)
    )
    parser.add_argument("--verify", action="store_true")
    parser.add_argument("--output", type=Path, required=True)
    args = parser.parse_args()
    if args.output.exists():
        parser.error("Use a fresh output path")
    shape = tuple(args.shape)
    config = (
        MODULE["Config"](**json.loads(args.config))
        if args.config
        else MODULE["CONFIGS_BY_SHAPE"][shape]
    )
    MODULE["validate_config"](config, shape)
    report = (
        verify(config, shape, args.seeds, args.timing)
        if args.verify
        else compare(config, shape, args.timing)
    )
    args.output.parent.mkdir(parents=True, exist_ok=True)
    args.output.write_text(json.dumps(report, indent=2) + "\n")


if __name__ == "__main__":
    main()
